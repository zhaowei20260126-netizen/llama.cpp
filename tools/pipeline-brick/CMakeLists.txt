set(TARGET llama-pipeline-brick)

add_executable(${TARGET} pipeline-brick.cpp)
target_link_libraries(${TARGET} PRIVATE llama-common llama ${CMAKE_THREAD_LIBS_INIT})
target_compile_features(${TARGET} PRIVATE cxx_std_17)

if(UNIX AND NOT APPLE)
    find_library(IBVERBS_LIB NAMES ibverbs libibverbs.so.1)
    find_path(IBVERBS_INCLUDE_DIR infiniband/verbs.h)
    if(IBVERBS_LIB AND IBVERBS_INCLUDE_DIR)
        target_compile_definitions(${TARGET} PRIVATE LLAMA_PIPELINE_BRICK_IBVERBS)
        target_include_directories(${TARGET} PRIVATE ${IBVERBS_INCLUDE_DIR})
        target_link_libraries(${TARGET} PRIVATE ${IBVERBS_LIB})
        message(STATUS "pipeline-brick: ib-rdma enabled via ${IBVERBS_LIB}")
    else()
        message(STATUS "pipeline-brick: ib-rdma disabled; libibverbs or infiniband/verbs.h not found")
    endif()
endif()

if(LLAMA_TOOLS_INSTALL)
    install(TARGETS ${TARGET} RUNTIME)
endif()
